 function polishMobileHeader(){
  const mobile=window.innerWidth<=700;
  const bn=document.querySelector('.brand-name'),tm=document.querySelector('.top-meta'),left=document.querySelector('.top-left'),who=document.querySelector('#who'),actions=who?.parentElement;
  if(bn){bn.dataset.fullText=bn.dataset.fullText||bn.textContent;if(mobile)bn.textContent='FONKON';else if(bn.dataset.fullText)bn.textContent=bn.dataset.fullText}
  if(tm){tm.dataset.fullText=tm.dataset.fullText||tm.textContent;if(mobile)tm.textContent='Market OS · 运营增强版 R3';else if(tm.dataset.fullText)tm.textContent=tm.dataset.fullText}
  const badge=document.querySelector('#releaseBadge');
  if(mobile&&badge&&left&&badge.parentElement!==left){left.appendChild(badge)}
  if(!mobile&&badge&&actions&&badge.parentElement!==actions){actions.insertBefore(badge,who||actions.firstChild)}
  const tools=document.querySelector('#mobileToolsBtn'),bell=document.querySelector('#reminderBell'),logout=document.querySelector('#logout');
  if(mobile&&actions){[tools,bell,logout].forEach(el=>{if(el&&el.parentElement===actions)actions.appendChild(el)})}
 }
