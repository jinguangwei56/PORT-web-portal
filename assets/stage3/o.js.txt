/* FONKON R9: stable admin action bindings */
(function(){
  async function audit(action,targetType,targetId,detail={}){try{await req('/rest/v1/admin_audit_log',{method:'POST',headers:{Prefer:'return=minimal'},body:JSON.stringify({actor_id:fonkonCurrentUid(),action,target_type:targetType,target_id:targetId||null,detail})})}catch(e){console.warn('audit skipped',e)}}
  window.copyInvite=async code=>{try{await navigator.clipboard.writeText(code);toast('邀请码已复制')}catch{prompt('请复制邀请码',code)}};
  window.makeSalesInvite=async()=>{try{const r=await edge({action:'invite',role:'sales'});await audit('create_invite','staff_invite',null,{code:r.code});toast('业务员邀请码已生成：'+r.code);await teamView();const box=$('#inviteResult');if(box){box.classList.remove('hide');box.innerHTML='<b style="font-size:18px">'+esc(r.code)+'</b> <button class="btn btn-soft" onclick="copyInvite(\''+r.code+'\')">复制</button><br><span class="muted">7天有效、一次性使用；注册后仍须最高权限审批。</span>'}}catch(e){toast('生成邀请码失败：'+e.message)}};
  window.approveSales=async id=>{if(!confirm('确认审批通过该业务员？'))return;try{await edge({action:'approve_user',user_id:id});await audit('approve_user','profile',id);toast('已审批通过');await teamView()}catch(e){toast('审批失败：'+e.message)}};
  window.rejectSales=async id=>{const note=prompt('请输入驳回原因','资料待核实');if(note===null)return;try{await edge({action:'reject_user',user_id:id,note});await audit('reject_user','profile',id,{note});toast('已驳回');await teamView()}catch(e){toast('驳回失败：'+e.message)}};
  window.toggleSales=async(id,active)=>{if(!confirm(active?'确认恢复该业务员账号？':'确认停用该业务员账号？'))return;try{await edge({action:'set_user_active',user_id:id,active});await audit(active?'enable_user':'disable_user','profile',id);toast(active?'账号已恢复':'账号已停用');await teamView()}catch(e){toast('账号操作失败：'+e.message)}};
  window.revokeInvite=async id=>{if(!confirm('确认撤销该邀请码？'))return;try{await req('/rest/v1/staff_invites?id=eq.'+encodeURIComponent(id),{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({active:false})});await audit('revoke_invite','staff_invite',id);toast('邀请码已撤销');await teamView()}catch(e){toast('撤销失败：'+e.message)}};
  window.adminResetPassword=async id=>{if(!confirm('确认重置该业务员密码？'))return;try{const r=await edge({action:'admin_reset_password',user_id:id});await audit('admin_reset_password','profile',id);if(r?.temporary_password)prompt('临时密码（请安全发送给业务员）',r.temporary_password);else toast('密码已重置')}catch(e){toast('重置密码失败：'+e.message)}};
  window.handoverSales=async from=>{try{await fonkonRefreshMyProfile();await refreshAll();const source=profiles.find(p=>p.id===from);const targets=profiles.filter(p=>p.role==='sales'&&p.id!==from&&p.active&&p.approval_status==='approved');const cs=customers.filter(c=>c.owner_id===from);if(!cs.length)return toast('该业务员没有客户需要交接');if(!targets.length)return toast('暂无其他已启用业务员可接收客户');modal('客户一键交接','<div class="notice"><b>'+esc(source?.name||'该业务员')+'</b> 名下 '+cs.length+' 个客户。</div><form id="r9handover" class="grid"><div class="field span12"><label>接收业务员</label><select name="to">'+targets.map(p=>'<option value="'+p.id+'">'+esc(p.name||p.email)+'</option>').join('')+'</select></div><div class="field span12"><label>交接原因</label><input name="reason" value="人员调整/客户交接"></div><div class="span12"><button class="btn btn-primary">确认交接</button></div></form>');$('#r9handover').onsubmit=async e=>{e.preventDefault();try{const x=Object.fromEntries(new FormData(e.target));const hist=cs.map(c=>({customer_id:c.id,from_owner_id:from,to_owner_id:x.to,changed_by:fonkonCurrentUid(),reason:x.reason||'客户交接'}));if(hist.length)await req('/rest/v1/customer_assignment_history',{method:'POST',headers:{Prefer:'return=minimal'},body:JSON.stringify(hist)});await req('/rest/v1/customers?owner_id=eq.'+encodeURIComponent(from),{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({owner_id:x.to})});await req('/rest/v1/customer_tasks?owner_id=eq.'+encodeURIComponent(from),{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({owner_id:x.to})});await audit('transfer_customers','profile',from,{to_user_id:x.to,count:cs.length,reason:x.reason});closeModal();toast('客户交接完成');await refreshAll();await teamView()}catch(z){toast('交接失败：'+z.message)}}}catch(e){toast('交接准备失败：'+e.message)}};
})();
