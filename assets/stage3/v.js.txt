/* FONKON R10.3 V: stable session guardian + final rich-page router */
(function(){
  const CORE={version:'R10.3-STABLE-SESSION2',refreshes:0,profileReads:0,errors:[]};
  window.FONKON_STABLE_CORE=CORE;
  const err=e=>String(e?.message||e?.error_description||e?.error||e||'未知错误').slice(0,320);
  const authish=e=>/401|403|jwt|token|expired|invalid login|refresh token|not authorized/i.test(err(e));
  const decode=token=>{try{let x=String(token||'').split('.')[1]||'';if(!x)return {};x=x.replace(/-/g,'+').replace(/_/g,'/');while(x.length%4)x+='=';return JSON.parse(atob(x))}catch{return {}}};
  let refreshFlight=null,heartbeatFlight=null;
  function readStored(){try{if(typeof window.fonkonReadStoredSession==='function')return window.fonkonReadStoredSession()}catch{}try{const raw=localStorage.getItem('fonkon_session');return raw?JSON.parse(raw):null}catch{return null}}
  const rank=x=>Number(x?.expires_at||decode(x?.access_token).exp||0);
  function persist(){if(ses?.access_token)saveSession(ses)}
  function hydrate(){const saved=readStored();if(saved?.access_token&&(!ses?.access_token||rank(saved)>rank(ses)))ses=saved;if(!ses?.access_token)return null;const p=decode(ses.access_token);if(!ses.user?.id&&p.sub)ses.user={id:p.sub,email:p.email||ses.user?.email||null};persist();return p}
  function terminalAuthError(message,status){const z=Error(message);z.status=status;z.fonkonTerminalAuth=([400,401].includes(Number(status))&&/invalid[_ ]?grant|invalid.*refresh|refresh.*(?:not found|revoked|expired)/i.test(message));return z}
  async function performRefresh(force,requestedToken){const p=hydrate();if(!ses?.refresh_token)throw terminalAuthError('当前会话缺少 refresh token',401);if(requestedToken&&ses.refresh_token!==requestedToken)return ses;const exp=Number(p?.exp||rank(ses)||0);if(!force&&exp&&exp*1000>Date.now()+120000)return ses;const used=ses.refresh_token;let r;try{r=await fetch(U+'/auth/v1/token?grant_type=refresh_token',{method:'POST',headers:{apikey:K,'content-type':'application/json'},body:JSON.stringify({refresh_token:used})})}catch(e){throw Error('网络暂时不可用，登录状态已保留：'+err(e))}const t=await r.text();let j={};try{j=t?JSON.parse(t):{}}catch{}if(!r.ok||!j?.access_token){const newer=readStored();if(newer?.refresh_token&&newer.refresh_token!==used){ses=newer;hydrate();return ses}throw terminalAuthError(j?.error_description||j?.message||j?.error||('HTTP '+r.status),r.status)}saveSession(j);hydrate();CORE.refreshes++;return ses}
  async function refreshToken(force=true){if(refreshFlight)return refreshFlight;const requested=readStored()?.refresh_token||ses?.refresh_token||null;const run=()=>performRefresh(force,requested);const locked=typeof navigator!=='undefined'&&navigator.locks?.request?navigator.locks.request('fonkon-auth-refresh',{mode:'exclusive'},run):run();refreshFlight=Promise.resolve(locked).finally(()=>{refreshFlight=null});return refreshFlight}
  async function ensureSession(forceRefresh=false){const p=hydrate();if(!ses?.access_token)throw terminalAuthError('当前没有可恢复的登录会话',401);const exp=Number(p?.exp||rank(ses)||0);if(forceRefresh||(exp&&exp*1000<Date.now()+120000))await refreshToken(!!forceRefresh);hydrate();if(!ses?.user?.id)throw terminalAuthError('登录会话缺少用户ID',401);return ses}
  async function ensureProfileStable(force=false){await ensureSession(false);if(!force&&profile?.id&&profile?.role)return profile;const uid=ses.user.id;try{const rows=await req('/rest/v1/profiles?id=eq.'+encodeURIComponent(uid)+'&select=*');if(!rows?.length)throw Error('当前账号没有Profile档案');profile=rows[0];CORE.profileReads++;persist()}catch(e){if(authish(e)&&ses?.refresh_token){await refreshToken(true);const rows=await req('/rest/v1/profiles?id=eq.'+encodeURIComponent(ses.user.id)+'&select=*');if(!rows?.length)throw Error('当前账号没有Profile档案');profile=rows[0];CORE.profileReads++}else throw e}const who=document.querySelector('#who');if(who)who.innerHTML='<b>'+esc(profile.name||profile.email||ses.user.email||'当前账号')+'</b> · '+(profile.role==='admin'?'最高权限':'业务员');document.querySelectorAll('[data-view="manager"],[data-view="team"],[data-view="settings"]').forEach(b=>b.style.display=profile.role==='admin'?'':'none');return profile}
  async function refreshBusinessData(){await ensureProfileStable(false);try{await refreshAll()}catch(e){if(authish(e)&&ses?.refresh_token){await refreshToken(true);await ensureProfileStable(true);await refreshAll()}else{CORE.errors.unshift({at:new Date().toISOString(),where:'refreshAll',message:err(e)});throw e}}}
  window.fonkonEnsureSession=ensureSession;window.fonkonEnsureProfile=ensureProfileStable;window.fonkonRefreshMyProfile=()=>ensureProfileStable(true);window.fonkonIsAdmin=()=>!!(profile?.role==='admin'&&profile?.active===true&&profile?.approval_status==='approved');window.fonkonRefreshBusinessData=refreshBusinessData;window.fonkonIsTerminalAuthError=e=>!!e?.fonkonTerminalAuth;window.fonkonHasStoredSession=()=>!!readStored();
  try{isManager=function(){return window.fonkonIsAdmin()};window.isManager=isManager}catch{}
  const oldRender=window.render;
  window.render=render=async function(view){try{await ensureProfileStable(false);return await oldRender(view)}catch(e){CORE.errors.unshift({at:new Date().toISOString(),where:'render:'+view,message:err(e)});CORE.errors=CORE.errors.slice(0,40);const box=document.querySelector('#view');if(box)box.innerHTML=pageHead(view==='knowledge'?'FONKON 水果知识中心':view==='manager'?'最高权限看板':view==='team'?'账号审批':view==='settings'?'评分规则':'页面','会话恢复失败')+'<div class="card card-pad"><div class="danger-note"><b>加载失败：</b>'+esc(err(e))+'</div><div class="muted" style="margin-top:8px">登录信息仍保留。请点击“重新连接”，系统会尝试续期，不会因为普通业务数据错误清空登录。</div><button class="btn btn-primary" style="margin-top:10px" id="reconnectSession">重新连接</button></div>';const b=document.querySelector('#reconnectSession');if(b)b.onclick=async()=>{try{await ensureSession(true);await ensureProfileStable(true);await refreshBusinessData();toast('会话已恢复');await window.render(view)}catch(z){toast('恢复失败：'+err(z))}}}}
  const login=document.querySelector('#loginForm');if(login)login.onsubmit=async e=>{e.preventDefault();const msg=document.querySelector('#authMsg');try{if(msg)msg.textContent='登录中…';const r=await fetch(U+'/auth/v1/token?grant_type=password',{method:'POST',headers:{apikey:K,'content-type':'application/json'},body:JSON.stringify({email:document.querySelector('#loginEmail').value,password:document.querySelector('#loginPassword').value})});const x=await r.json();if(!r.ok)throw Error(x?.error_description||x?.message||x?.error||'登录失败');saveSession(x);hydrate();await ensureProfileStable(true);if(profile.approval_status==='pending'){clearSession();if(msg)msg.textContent='注册申请正在等待最高权限审批。';return}if(profile.approval_status==='rejected'){const m='注册申请未通过'+(profile.approval_note?'：'+profile.approval_note:'。请联系最高权限。');clearSession();if(msg)msg.textContent=m;return}if(!profile.active){clearSession();if(msg)msg.textContent='账号当前已停用，请联系最高权限。';return}try{await refreshAll()}catch(z){console.warn('[FONKON] 登录后业务数据加载降级',z);if(typeof toast==='function')toast('已登录，部分数据稍后自动重试')}if(msg)msg.textContent='';showApp()}catch(z){if(msg)msg.textContent=err(z)}};
  const logout=document.querySelector('#logout');if(logout)logout.onclick=()=>{clearSession();showApp()};
  const oldShow=window.showApp;window.showApp=function(){if((!ses||!profile)&&readStored()?.access_token){ensureProfileStable(false).then(()=>{oldShow();try{if(typeof updateReminderBell==='function')updateReminderBell()}catch{}}).catch(()=>oldShow());return}return oldShow()};
  async function heartbeat(reason='heartbeat'){if(document.visibilityState==='hidden')return;if(heartbeatFlight)return heartbeatFlight;const saved=readStored();if(!saved?.access_token&&!ses?.access_token)return;heartbeatFlight=(async()=>{try{await ensureSession(false);if(!profile?.id)await ensureProfileStable(false)}catch(e){CORE.errors.unshift({at:new Date().toISOString(),where:reason,message:err(e)});CORE.errors=CORE.errors.slice(0,40)}})().finally(()=>{heartbeatFlight=null});return heartbeatFlight}
  let heartbeatTimer=null;const queueHeartbeat=reason=>{clearTimeout(heartbeatTimer);heartbeatTimer=setTimeout(()=>heartbeat(reason),180)};
  document.addEventListener('visibilitychange',()=>{if(document.visibilityState==='visible')queueHeartbeat('visibility')});window.addEventListener('pageshow',()=>queueHeartbeat('pageshow'));window.addEventListener('focus',()=>queueHeartbeat('focus'));window.addEventListener('online',()=>queueHeartbeat('online'));window.addEventListener('storage',e=>{if(e.key==='fonkon_session'){hydrate();queueHeartbeat('storage')}});setInterval(()=>heartbeat('timer'),5*60*1000);
  function mark(){try{const tm=document.querySelector('.top-meta');if(tm)tm.textContent='1.2 市场作战体系版 · R10.3 稳定整合';let b=document.querySelector('#releaseBadge'),who=document.querySelector('#who');if(!b&&who){b=document.createElement('span');b.id='releaseBadge';b.className='badge b-green';who.parentElement.insertBefore(b,who)}if(b)b.textContent='R10.3'}catch{}}
  [80,500,1500,3500,7000].forEach(ms=>setTimeout(mark,ms));setTimeout(()=>{if(readStored()?.access_token){ensureProfileStable(false).then(()=>{try{oldShow()}catch{}}).catch(()=>{})}},120);
})();
