/* Knowledge Center: admin review + standard knowledge maintenance */
(function(){
const oldOpen=window.openFruitKnowledge;
window.openFruitKnowledge=async function(fruit){
 await oldOpen(fruit);
 const notes=await req('/rest/v1/knowledge_submissions?fruit=eq.'+encodeURIComponent(fruit)+'&review_status=eq.'+encodeURIComponent('已采纳')+'&select=*&order=reviewed_at.desc&limit=30');
 if(notes.length){const box='<div class="knowledge-block"><h3>📚 公司已采纳补充 / 纠错</h3>'+notes.map(x=>'<div class="knowledge-row"><b>'+esc(x.submission_type)+'｜'+esc(x.title)+'</b><div>'+esc(x.content)+'</div><div class="muted">来源：'+esc(x.source_text||'-')+'｜证据 '+esc(x.evidence_level)+'｜审核 '+dOnly(x.reviewed_at)+'</div></div>').join('')+'</div>';$('#view').insertAdjacentHTML('beforeend',box)}
 if(profile?.role==='admin')$('#view').insertAdjacentHTML('beforeend','<div class="knowledge-block"><h3>🛠 最高权限维护</h3><div class="k-actions"><button class="btn btn-primary" onclick="standardKnowledgeModal(\'origin\',\''+esc(fruit)+'\')">＋产区产季</button><button class="btn btn-primary" onclick="standardKnowledgeModal(\'clearance\',\''+esc(fruit)+'\')">＋通关检测</button><button class="btn btn-primary" onclick="standardKnowledgeModal(\'coldchain\',\''+esc(fruit)+'\')">＋冷链品质</button></div><div class="muted" style="margin-top:8px">这里只写入已核验的公司标准知识；业务员口述先走“市场情报/知识补充”审核。</div></div>');
}
window.knowledgeReviewView=async function(){
 if(profile?.role!=='admin')return toast('仅最高权限可审核');
 const [intel,subs]=await Promise.all([req('/rest/v1/market_intel?review_status=eq.'+encodeURIComponent('待审核')+'&select=*&order=collected_at.asc'),req('/rest/v1/knowledge_submissions?review_status=eq.'+encodeURIComponent('待审核')+'&select=*&order=created_at.asc')]);
 let html=pageHead('知识审核台','业务员原始情报先审核，避免未经验证的信息污染公司知识。','<button class="btn btn-soft" onclick="knowledgeCenterView(\'fruits\')">← 知识中心</button>');
 const intelRows=intel.length?intel.map(x=>'<div class="knowledge-row intel-pending"><b>'+esc(x.fruit)+'｜'+esc(x.intel_type)+'</b><div>'+esc(x.finding||x.notes||'-')+'</div><div class="muted">'+esc(x.market_name||x.city||'-')+' · '+dOnly(x.collected_at)+' · 来源 '+esc(x.source_name||'-')+' · 证据 '+esc(x.evidence_level)+'</div>'+(x.customer_quote?'<div class="muted">原话：'+esc(x.customer_quote)+'</div>':'')+'<div class="k-actions" style="margin-top:8px"><button class="btn btn-primary" onclick="reviewIntel(\''+x.id+'\',true)">采纳为共享情报</button><button class="btn btn-soft" onclick="reviewIntel(\''+x.id+'\',false)">驳回</button></div></div>').join(''):'<div class="k-empty">暂无待审核市场情报</div>';
 const subRows=subs.length?subs.map(x=>'<div class="knowledge-row intel-pending"><b>'+esc(x.fruit||'通用')+'｜'+esc(x.submission_type)+'｜'+esc(x.title)+'</b><div>'+esc(x.content)+'</div><div class="muted">来源 '+esc(x.source_text||'-')+' · 证据 '+esc(x.evidence_level)+'</div>'+(x.customer_quote?'<div class="muted">原话：'+esc(x.customer_quote)+'</div>':'')+'<div class="k-actions" style="margin-top:8px"><button class="btn btn-primary" onclick="reviewSubmission(\''+x.id+'\',true)">采纳为共享补充</button><button class="btn btn-soft" onclick="reviewSubmission(\''+x.id+'\',false)">驳回</button></div></div>').join(''):'<div class="k-empty">暂无待审核知识补充</div>';
 html+='<div class="two"><div class="knowledge-block"><h3>市场情报待审核 '+intel.length+'</h3>'+intelRows+'</div><div class="knowledge-block"><h3>知识补充待审核 '+subs.length+'</h3>'+subRows+'</div></div>';$('#view').innerHTML=html;
}
window.reviewIntel=async function(id,ok){const note=prompt(ok?'审核备注（可留空）':'驳回原因（建议填写）','')??'';await req('/rest/v1/market_intel?id=eq.'+id,{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({review_status:ok?'已采纳':'已驳回',reviewed_by:ses.user.id,reviewed_at:new Date().toISOString(),review_note:note})});toast(ok?'已采纳为公司共享情报':'已驳回');knowledgeReviewView()}
window.reviewSubmission=async function(id,ok){const note=prompt(ok?'审核备注（可留空）':'驳回原因（建议填写）','')??'';await req('/rest/v1/knowledge_submissions?id=eq.'+id,{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({review_status:ok?'已采纳':'已驳回',reviewed_by:ses.user.id,reviewed_at:new Date().toISOString(),review_note:note})});toast(ok?'已采纳为公司共享补充':'已驳回');knowledgeReviewView()}
window.standardKnowledgeModal=function(kind,fruit){
 if(profile?.role!=='admin')return toast('仅最高权限可维护标准知识');
 let fields='';
 if(kind==='origin')fields=input('country','国家*','','text','span4','required')+input('region','产区','','text','span4')+input('harvest_start_month','采摘开始月','','number')+input('harvest_end_month','采摘结束月','','number')+input('peak_start_month','高峰开始月','','number')+input('peak_end_month','高峰结束月','','number')+input('china_import_start_month','对华进口开始月','','number')+input('china_import_end_month','对华进口结束月','','number')+input('annual_output_tons','年产量/吨','','number')+input('export_to_china_tons','对华出口/吨','','number')+'<div class="field span12"><label>说明</label><textarea name="notes"></textarea></div>';

 if(kind==='coldchain')fields=input('origin_country','来源国（空=通用）','','text','span4')+input('temp_min_c','最低温℃','','number','span4','step="0.1"')+input('temp_max_c','最高温℃','','number','span4','step="0.1"')+input('humidity_min_pct','最低湿度%','','number')+input('humidity_max_pct','最高湿度%','','number')+input('shelf_life_days_min','货架期最短天','','number')+input('shelf_life_days_max','货架期最长天','','number')+input('precooling','预冷要求','','text','span6')+input('maturity_guidance','成熟度/采摘要求','','text','span6')+'<div class="field span6"><label>乙烯/气调</label><textarea name="ethylene_notes"></textarea></div><div class="field span6"><label>常见货损风险</label><textarea name="common_damage_risks"></textarea></div><div class="field span12"><label>运输备注</label><textarea name="transport_notes"></textarea></div>';
 modal('维护标准知识｜'+fruit,'<form id="stdKForm" class="grid"><div class="field span4"><label>水果</label><input value="'+esc(fruit)+'" disabled></div>'+fields+input('source_text','来源/依据*','','text','span6','required')+input('source_date','来源日期','','date','span3')+select('evidence_level','证据等级',['A','B','C','D','未知'],'A','span3')+input('last_verified_at','最后核验日期','','date','span3')+'<div class="span12 notice">标准知识必须可追溯。A/B级优先；C/D级建议继续核验，不要把经验值包装成全国固定规则。</div><div class="span12"><button class="btn btn-primary">保存标准知识</button></div></form>');
 $('#stdKForm').onsubmit=async e=>{e.preventDefault();try{const x=Object.fromEntries(new FormData(e.target));x.fruit=fruit;x.created_by=ses.user.id;if(kind==='clearance'){x.failure_outcomes=(x.failure_outcomes_text||'').split('/').map(s=>s.trim()).filter(Boolean);delete x.failure_outcomes_text}Object.keys(x).forEach(k=>{if(['harvest_start_month','harvest_end_month','peak_start_month','peak_end_month','china_import_start_month','china_import_end_month','annual_output_tons','export_to_china_tons','typical_min_days','typical_max_days','temp_min_c','temp_max_c','humidity_min_pct','humidity_max_pct','shelf_life_days_min','shelf_life_days_max'].includes(k))x[k]=x[k]===''?null:+x[k]});const table=kind==='origin'?'fruit_origin_knowledge':kind==='clearance'?'fruit_clearance_knowledge':'fruit_coldchain_knowledge';await req('/rest/v1/'+table,{method:'POST',headers:{Prefer:'return=minimal'},body:JSON.stringify(x)});closeModal();toast('标准知识已保存');openFruitKnowledge(fruit)}catch(z){toast(z.message)}};
}
window.knowledgeView=window.knowledgeCenterView;
setTimeout(()=>{const b=document.querySelector('.nav-btn[data-view="knowledge"] span:last-child');if(b)b.textContent='知识中心';},0);
