(function(){
const oldTodayDashboard=typeof todayDashboard==='function'?todayDashboard:null;
const fmtDate=x=>x?new Date(x).toLocaleString('zh-CN',{month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit'}):'-';
const statusText=p=>p.approval_status==='pending'?'待审批':p.approval_status==='rejected'?'已驳回':p.active?'已启用':'已停用';
const statusBadge=p=>{const s=statusText(p);const c=s==='待审批'?'b-orange':s==='已启用'?'b-green':s==='已驳回'?'b-red':'b-gray';return '<span class="badge '+c+'">'+s+'</span>'};

const regBtn=document.querySelector('#registerForm button');if(regBtn)regBtn.textContent='提交注册申请';
const regCode=document.querySelector('#regCode');if(regCode){const f=regCode.closest('.field');if(f&&!f.querySelector('.approval-hint'))f.insertAdjacentHTML('beforeend','<div class="muted approval-hint" style="margin-top:6px">业务员注册后须由最高权限审批，通过后才可进入系统。</div>')}
const teamNav=document.querySelector('.nav-btn[data-view="team"] span:last-child');if(teamNav)teamNav.textContent='账号审批';

$('#registerForm').onsubmit=async e=>{e.preventDefault();try{$('#authMsg').textContent='正在提交注册申请…';const r=await edge({action:'register',name:$('#regName').value,email:$('#regEmail').value,password:$('#regPassword').value,code:$('#regCode').value});if(r.pending){$('#authMsg').textContent='申请已提交，等待最高权限审批。审批通过后再使用该邮箱和密码登录。';e.target.reset()}else{$('#authMsg').textContent='最高权限账号创建成功，请登录';$('#tabLogin').click();$('#loginEmail').value=$('#regEmail').value}}catch(z){$('#authMsg').textContent=z.message}};

$('#loginForm').onsubmit=async e=>{e.preventDefault();try{$('#authMsg').textContent='登录中…';const x=await req('/auth/v1/token?grant_type=password',{method:'POST',body:JSON.stringify({email:$('#loginEmail').value,password:$('#loginPassword').value})});saveSession(x);await loadProfile();if(profile.approval_status==='pending'){const m='注册申请正在等待最高权限审批，审批通过后即可进入系统。';clearSession();$('#authMsg').textContent=m;return}if(profile.approval_status==='rejected'){const m='注册申请未通过'+(profile.approval_note?'：'+profile.approval_note:'。请联系最高权限。');clearSession();$('#authMsg').textContent=m;return}if(!profile.active){clearSession();$('#authMsg').textContent='账号当前已停用，请联系最高权限。';return}await refreshAll();$('#authMsg').textContent='';showApp()}catch(z){clearSession();$('#authMsg').textContent=z.message}};

async function adminProfiles(){profiles=await req('/rest/v1/profiles?select=*&order=applied_at.desc.nullslast,name.asc');return profiles}
async function adminInvites(){return await req('/rest/v1/staff_invites?select=*&order=created_at.desc&limit=50')}
async function adminAudits(){return await req('/rest/v1/admin_audit_log?select=*&order=created_at.desc&limit=30')}
async function writeAudit(action,targetType,targetId,detail={}){try{await req('/rest/v1/admin_audit_log',{method:'POST',headers:{Prefer:'return=minimal'},body:JSON.stringify({actor_id:ses.user.id,action,target_type:targetType,target_id:targetId||null,detail})})}catch{}}
function actionName(a){return ({approve_user:'审批通过',reject_user:'驳回申请',enable_user:'恢复账号',disable_user:'停用账号',create_invite:'生成邀请码',revoke_invite:'撤销邀请码',transfer_customers:'客户交接'}[a]||a)}
function profileRow(p){const count=customers.filter(c=>c.owner_id===p.id).length;return '<div class="rank" style="align-items:flex-start"><div class="rank-no">'+esc((p.name||'?').slice(0,1))+'</div><div class="rank-main"><b>'+esc(p.name||'未命名')+'</b> '+statusBadge(p)+'<div>'+esc(p.email||'-')+' · 业务员 · 客户 '+count+' 个</div><div class="muted">申请 '+fmtDate(p.applied_at)+' · 审批 '+fmtDate(p.reviewed_at)+(p.approval_note?' · '+esc(p.approval_note):'')+'</div></div><div class="actions">'+(p.approval_status==='pending'?'<button class="btn btn-primary" onclick="approveSales(\''+p.id+'\')">通过</button><button class="btn btn-soft" onclick="rejectSales(\''+p.id+'\')">驳回</button>':p.approval_status==='approved'?(p.active?'<button class="btn btn-soft" onclick="toggleSales(\''+p.id+'\',false)">停用</button>':'<button class="btn btn-primary" onclick="toggleSales(\''+p.id+'\',true)">恢复</button>'):'')+(count?'<button class="btn btn-soft" onclick="handoverSales(\''+p.id+'\')">交接客户</button>':'')+'</div></div>'}

window.approveSales=async id=>{if(!confirm('确认审批通过该业务员？通过后可立即登录并只访问自己的业务数据。'))return;try{await edge({action:'approve_user',user_id:id});await writeAudit('approve_user','profile',id);await adminProfiles();toast('已审批通过');teamView()}catch(e){toast(e.message)}};
window.rejectSales=async id=>{const note=prompt('请输入驳回原因（业务员登录时会看到）','资料待核实');if(note===null)return;try{await edge({action:'reject_user',user_id:id,note});await writeAudit('reject_user','profile',id,{note});await adminProfiles();toast('已驳回申请');teamView()}catch(e){toast(e.message)}};
window.toggleSales=async(id,active)=>{if(!confirm(active?'确认恢复该业务员账号？':'确认停用该业务员？停用后其现有登录立即失去业务数据访问权限。'))return;try{await edge({action:'set_user_active',user_id:id,active});await writeAudit(active?'enable_user':'disable_user','profile',id);await adminProfiles();toast(active?'账号已恢复':'账号已停用');teamView()}catch(e){toast(e.message)}};
window.makeSalesInvite=async()=>{try{const r=await edge({action:'invite',role:'sales'});await writeAudit('create_invite','staff_invite',null,{code:r.code});const box=$('#inviteResult');box.classList.remove('hide');box.innerHTML='<b>'+esc(r.code)+'</b><br><span class="muted">7天有效、一次性使用；注册后仍须最高权限审批。</span>';await teamView()}catch(e){toast(e.message)}};
window.revokeInvite=async id=>{if(!confirm('确认撤销该邀请码？'))return;try{await req('/rest/v1/staff_invites?id=eq.'+id,{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({active:false})});await writeAudit('revoke_invite','staff_invite',id);toast('邀请码已撤销');teamView()}catch(e){toast(e.message)}};
window.handoverSales=async from=>{const source=profiles.find(p=>p.id===from);const targets=profiles.filter(p=>p.role==='sales'&&p.id!==from&&p.active&&p.approval_status==='approved');const count=customers.filter(c=>c.owner_id===from).length;if(!count)return toast('该业务员没有客户需要交接');if(!targets.length)return toast('暂无其他已启用业务员可接收客户');modal('客户一键交接','<div class="notice"><b>'+esc(source?.name||'该业务员')+'</b> 当前名下 '+count+' 个客户。交接后，客户、商机和拜访历史保留；后续任务负责人同步转移。</div><form id="handoverForm" class="grid"><div class="field span12"><label>接收业务员</label><select name="to">'+targets.map(p=>'<option value="'+p.id+'">'+esc(p.name||p.email)+'</option>').join('')+'</select></div><div class="field span12"><label>交接原因</label><input name="reason" value="人员调整/客户交接"></div><div class="span12"><button class="btn btn-primary">确认交接 '+count+' 个客户</button></div></form>');$('#handoverForm').onsubmit=async e=>{e.preventDefault();try{const x=Object.fromEntries(new FormData(e.target));const cs=customers.filter(c=>c.owner_id===from);const hist=cs.map(c=>({customer_id:c.id,from_owner_id:from,to_owner_id:x.to,changed_by:ses.user.id,reason:x.reason||'客户交接'}));if(hist.length)await req('/rest/v1/customer_assignment_history',{method:'POST',headers:{Prefer:'return=minimal'},body:JSON.stringify(hist)});await req('/rest/v1/customers?owner_id=eq.'+from,{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({owner_id:x.to})});await req('/rest/v1/customer_tasks?owner_id=eq.'+from,{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({owner_id:x.to})});await writeAudit('transfer_customers','profile',from,{to_user_id:x.to,count:cs.length,reason:x.reason});await refreshAll();await adminProfiles();closeModal();toast('已完成客户交接');teamView()}catch(z){toast(z.message)}}};

teamView=async function(){if(profile?.role!=='admin'){$('#view').innerHTML=pageHead('账号审批','仅最高权限可进入。')+'<div class="card empty">无权限</div>';return}try{const [ps,invs,aud]=await Promise.all([adminProfiles(),adminInvites(),adminAudits()]);const pending=ps.filter(p=>p.role==='sales'&&p.approval_status==='pending');const enabled=ps.filter(p=>p.role==='sales'&&p.approval_status==='approved'&&p.active);const stopped=ps.filter(p=>p.role==='sales'&&(p.approval_status==='rejected'||(p.approval_status==='approved'&&!p.active)));const now=Date.now();let html=pageHead('账号审批与团队控制','最高权限审批注册、控制账号、管理邀请码并处理客户交接。','<button class="btn btn-primary" onclick="makeSalesInvite()">＋生成业务员邀请码</button>');html+='<div class="kpis"><div class="card kpi"><div class="kpi-label">待审批</div><div class="kpi-value">'+pending.length+'</div><div class="kpi-foot">注册申请</div></div><div class="card kpi"><div class="kpi-label">启用业务员</div><div class="kpi-value">'+enabled.length+'</div><div class="kpi-foot">可正常登录</div></div><div class="card kpi"><div class="kpi-label">停用/驳回</div><div class="kpi-value">'+stopped.length+'</div><div class="kpi-foot">无业务访问权</div></div><div class="card kpi"><div class="kpi-label">有效邀请码</div><div class="kpi-value">'+invs.filter(i=>i.active&&!i.used_by&&(!i.expires_at||new Date(i.expires_at).getTime()>now)).length+'</div><div class="kpi-foot">7天一次性</div></div></div>';
html+='<div class="card card-pad section"><div class="section-title"><h2>🔔 待审批注册</h2><span class="hint">业务员未通过审批前无法访问客户数据</span></div>'+(pending.length?pending.map(profileRow).join(''):'<div class="empty">暂无待审批申请</div>')+'</div>';
html+='<div class="two section"><div class="card card-pad"><div class="section-title"><h2>✅ 已启用业务员</h2><span class="hint">只能查看自己名下数据</span></div>'+(enabled.length?enabled.map(profileRow).join(''):'<div class="empty">暂无业务员</div>')+'</div><div class="card card-pad"><div class="section-title"><h2>⛔ 停用 / 驳回</h2><span class="hint">可恢复已审批但停用的账号</span></div>'+(stopped.length?stopped.map(profileRow).join(''):'<div class="empty">暂无</div>')+'</div></div>';
html+='<div class="two section"><div class="card card-pad"><div class="section-title"><h2>邀请码</h2><button class="btn btn-primary" onclick="makeSalesInvite()">生成</button></div><div id="inviteResult" class="notice hide"></div>'+invs.slice(0,12).map(i=>{const valid=i.active&&!i.used_by&&(!i.expires_at||new Date(i.expires_at).getTime()>now);return '<div class="task-card"><div class="task-top"><b>'+esc(i.code)+'</b><span class="badge '+(valid?'b-green':'b-gray')+'">'+(i.used_by?'已使用':valid?'有效':'失效')+'</span></div><div class="task-meta">创建 '+fmtDate(i.created_at)+' · 到期 '+fmtDate(i.expires_at)+'</div>'+(valid?'<div style="margin-top:7px"><button class="btn btn-soft" onclick="revokeInvite(\''+i.id+'\')">撤销</button></div>':'')+'</div>'}).join('')+'</div><div class="card card-pad"><div class="section-title"><h2>管理员操作日志</h2><span class="hint">审批、停用、邀请码、客户交接留痕</span></div>'+aud.map(a=>'<div class="timeline-item"><div class="timeline-date">'+fmtDate(a.created_at)+' · '+esc(actionName(a.action))+'</div><div class="timeline-text">'+esc(a.target_type||'-')+(a.detail&&Object.keys(a.detail).length?' · '+esc(JSON.stringify(a.detail)):'')+'</div></div>').join('')+(aud.length?'':'<div class="empty">暂无操作记录</div>')+'</div></div>';
$('#view').innerHTML=html}catch(e){$('#view').innerHTML=pageHead('账号审批','加载失败')+'<div class="card empty">'+esc(e.message)+'</div>'}}
