 async function freshProfile(){
  if(!ses?.user?.id) throw Error('登录状态已失效，请重新登录');
  const a=await req('/rest/v1/profiles?id=eq.'+encodeURIComponent(ses.user.id)+'&select=*');
  if(!a.length) throw Error('未找到当前账号权限档案');
  profile=a[0];
  return profile;
 }
