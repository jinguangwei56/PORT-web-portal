function isManager(){return profile?.role==='admin'}
function showApp(){
  if(!ses||!profile){$('#auth').classList.remove('hide');$('#app').classList.add('hide');return}
  if(!profile.active){clearSession();$('#auth').classList.remove('hide');$('#app').classList.add('hide');$('#authMsg').textContent='账号已被最高权限停用';return}
  $('#auth').classList.add('hide');$('#app').classList.remove('hide');
  $('#who').innerHTML='<b>'+esc(profile.name||ses.user.email)+'</b> · '+(profile.role==='admin'?'最高权限':'业务员');
  $$('[data-view="manager"],[data-view="team"],[data-view="settings"]').forEach(b=>b.style.display=isManager()?'':'none');
  const mb=document.querySelector('[data-view="manager"] span:last-child'); if(mb)mb.textContent='最高权限看板';
  render('dashboard');
}

async function toggleSalesActive(id,newActive){
  try{
    await req('/rest/v1/rpc/fonkon_admin_set_sales_active',{method:'POST',body:JSON.stringify({target_user:id,new_active:newActive})});
    await refreshAll();toast(newActive?'业务员账号已恢复':'业务员账号已停用');teamView();
  }catch(e){toast(e.message)}
}

  const reps=profiles.filter(p=>p.role==='sales').map(p=>{const cs=customers.filter(c=>c.owner_id===p.id),ops=opportunities.filter(o=>cs.some(c=>c.id===o.customer_id)),vs=visits.filter(v=>v.created_by===p.id);return {name:p.name||'未命名',customers:cs.length,high:cs.filter(c=>['S','A'].includes(c.cv_grade)).length,red:ops.filter(o=>o.alert_level==='红').length,trial:ops.reduce((s,o)=>s+num(o.trial_volume),0),visits:vs.length,score:ops.reduce((s,o)=>s+num(o.final_priority),0)}}).sort((a,b)=>b.score-a.score);
