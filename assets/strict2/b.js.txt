function managerView(){
  if(!isManager()){$('#view').innerHTML=pageHead('最高权限看板','仅最高权限账号可查看全公司数据。')+'<div class="card empty">当前账号无最高权限</div>';return}
  const stages=['待验证','需求确认','方案','报价','试柜','持续合作'];
  const funnel=stages.map(s=>({s,n:opportunities.filter(o=>o.stage===s).length}));
  const reps=profiles.filter(p=>p.role==='sales').map(p=>{const cs=customers.filter(c=>c.owner_id===p.id),ops=opportunities.filter(o=>cs.some(c=>c.id===o.customer_id)),vs=visits.filter(v=>v.created_by===p.id);return {name:p.name||'未命名',customers:cs.length,high:cs.filter(c=>['S','A'].includes(c.cv_grade)).length,red:ops.filter(o=>o.alert_level==='红').length,trial:ops.reduce((s,o)=>s+num(o.trial_volume),0),visits:vs.length,score:ops.reduce((s,o)=>s+num(o.final_priority),0)}}).sort((a,b)=>b.score-a.score);
  const fruitMap={};opportunities.forEach(o=>{const k=o.fruit||'未知';fruitMap[k]??={fruit:k,count:0,switchable:0,sum:0};fruitMap[k].count++;fruitMap[k].switchable+=num(o.switchable_volume);fruitMap[k].sum+=num(o.final_priority)});
  const fruits=Object.values(fruitMap).map(x=>({...x,avg:x.count?x.sum/x.count:0})).sort((a,b)=>b.avg-a.avg).slice(0,8);
  let html=pageHead('最高权限作战看板','看全公司漏斗、业务员转化和试柜结果，不用“感觉不错”管理市场。','<button class="btn btn-soft" onclick="exportCustomers()">导出客户</button><button class="btn btn-soft" onclick="exportOpps()">导出商机</button>');
  html+='<div class="kpis"><div class="card kpi"><div class="kpi-label">团队客户</div><div class="kpi-value">'+customers.length+'</div><div class="kpi-foot">全公司客户池</div></div><div class="card kpi"><div class="kpi-label">有效商机</div><div class="kpi-value">'+opportunities.filter(o=>o.stage!=='关闭').length+'</div><div class="kpi-foot">未关闭</div></div><div class="card kpi"><div class="kpi-label">试柜柜量</div><div class="kpi-value">'+fmt(opportunities.reduce((s,o)=>s+num(o.trial_volume),0))+'</div><div class="kpi-foot">首批验证</div></div><div class="card kpi"><div class="kpi-label">持续合作</div><div class="kpi-value">'+opportunities.filter(o=>o.stage==='持续合作').length+'</div><div class="kpi-foot">进入复购</div></div></div>';
  html+='<div class="card card-pad"><div class="section-title"><h2>市场开发漏斗</h2><span class="hint">待验证→需求→方案→报价→试柜→持续合作</span></div><div class="funnel">'+funnel.map(x=>'<div class="funnel-step"><b>'+x.n+'</b><span>'+x.s+'</span></div>').join('')+'</div></div>';
  html+='<div class="two section"><div class="card card-pad"><div class="section-title"><h2>业务员作战排行</h2><span class="hint">看有效商机、拜访和试柜</span></div>'+reps.map((r,i)=>'<div class="rank"><div class="rank-no">'+(i+1)+'</div><div class="rank-main"><b>'+esc(r.name)+'</b><div>'+r.customers+'客户 · '+r.high+' A/S · '+r.red+'红灯 · '+r.visits+'拜访</div></div><div class="rank-val">'+fmt(r.trial)+'柜</div></div>').join('')+'</div><div class="card card-pad"><div class="section-title"><h2>水果机会排行</h2><span class="hint">平均作战优先分</span></div>'+fruits.map((f,i)=>'<div class="rank"><div class="rank-no">'+(i+1)+'</div><div class="rank-main"><b>'+esc(f.fruit)+'</b><div>'+f.count+'个商机 · 可切换 '+fmt(f.switchable)+'柜</div></div><div class="rank-val">'+f.avg.toFixed(1)+'</div></div>').join('')+'</div></div>';
  $('#view').innerHTML=html;
}
