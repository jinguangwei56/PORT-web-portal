async function changePassword(newPassword){
 if(!ses?.access_token)throw Error('请先登录');
 const r=await fetch(U+'/auth/v1/user',{method:'PUT',headers:{apikey:K,authorization:'Bearer '+ses.access_token,'content-type':'application/json'},body:JSON.stringify({password:newPassword})});
 const j=await r.json();if(!r.ok)throw Error(j?.message||j?.error_description||'修改密码失败');
 try{await req('/rest/v1/profiles?id=eq.'+ses.user.id,{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({force_password_change:false})});profile.force_password_change=false}catch{}
 return j;
}
