/* FONKON R11 P1: server-side customer/opportunity pagination. */
(function(){
  'use strict';
  const SIZE=50,state={customers:1,opportunities:1,customerTotal:0,oppTotal:0};
  const h=()=>{const x={apikey:K,'content-type':'application/json',Prefer:'count=exact'};if(ses?.access_token)x.authorization='Bearer '+ses.access_token;return x};
  async function ranged(path){const r=await fetch(U+path,{headers:h(),cache:'no-store'}),txt=await r.text();let data=[];try{data=txt?JSON.parse(txt):[]}catch{}if(!r.ok)throw Error(data?.message||data?.error||('HTTP '+r.status));const cr=r.headers.get('content-range')||'',m=cr.match(/\/(\d+|\*)$/),total=m&&m[1]!=='*'?Number(m[1]):data.length;return {data:Array.isArray(data)?data:[],total:Number.isFinite(total)?total:data.length}}
  const mergeById=(base,rows)=>{const m=new Map((base||[]).map(x=>[x.id,x]));rows.forEach(x=>m.set(x.id,x));return Array.from(m.values())};
  function pagers(kind,total){const pages=Math.max(1,Math.ceil(total/SIZE)),p=Math.min(state[kind],pages);state[kind]=p;return '<div class="actions" style="justify-content:flex-end;margin-top:10px;align-items:center"><span class="muted">共 '+total+' 条 · 第 '+p+'/'+pages+' 页 · 服务端每页 '+SIZE+' 条</span><button class="btn btn-soft" '+(p<=1?'disabled':'')+' onclick="fonkonR11ServerPage(\''+kind+'\',-1)">上一页</button><button class="btn btn-soft" '+(p>=pages?'disabled':'')+' onclick="fonkonR11ServerPage(\''+kind+'\',1)">下一页</button></div>'}
  const val=id=>document.querySelector('#'+id)?.value||'';
  function customerPath(){const parts=['archived_at=is.null','select=*','order=updated_at.desc','limit='+SIZE,'offset='+((state.customers-1)*SIZE)],q=val('cq').trim(),g=val('cg'),t=val('ct'),cc=val('cc');if(q){const or='(company_name.ilike.*'+q+'*,contact_name.ilike.*'+q+'*,primary_fruit.ilike.*'+q+'*,market_name.ilike.*'+q+'*,current_ports.ilike.*'+q+'*)';parts.push('or='+encodeURIComponent(or))}if(g)parts.push('cv_grade=eq.'+encodeURIComponent(g));if(t)parts.push('customer_type=eq.'+encodeURIComponent(t));if(cc)parts.push('primary_cargo_group=eq.'+encodeURIComponent(cc));return '/rest/v1/customers?'+parts.join('&')}
  function customerRows(rows){return '<div class="table-wrap"><table class="table"><thead><tr><th>客户</th><th>角色/决策</th><th>水果</th><th>年柜量</th><th>CV</th><th>DC</th><th>市场/口岸</th><th>下次跟进</th><th>负责人</th></tr></thead><tbody>'+rows.map(c=>'<tr><td><div class="link" onclick="openCustomer(\''+c.id+'\')">'+esc(c.company_name)+'</div><div class="muted">'+esc(c.contact_name||'')+' '+esc(c.phone||'')+'</div></td><td>'+esc(c.customer_type||'-')+'<div class="muted">'+esc(c.decision_role||'待确认')+' / DP '+fmt(c.decision_power)+'</div></td><td>'+esc(c.primary_fruit||c.core_fruits||'-')+' '+(c.primary_cargo_group?'<span class="chip">'+esc(c.primary_cargo_group)+'</span>':'')+'</td><td>'+fmt(c.annual_containers)+'</td><td>'+badge(c.cv_grade)+' <b>'+fmt(c.cv_score)+'</b></td><td><b>'+fmt(c.dc_score)+'%</b>'+progress(c.dc_score)+'</td><td>'+esc(c.market_name||'-')+'<div class="muted">'+esc(c.current_ports||'-')+'</div></td><td>'+dt(c.next_followup_at)+'</td><td>'+esc(ownerName(c.owner_id))+'</td></tr>').join('')+'</tbody></table></div>'}
  async function loadCustomers(){const root=document.querySelector('#customerTable');if(!root)return;root.innerHTML='<div class="empty">正在读取客户第 '+state.customers+' 页…</div>';try{const r=await ranged(customerPath());state.customerTotal=r.total;customers=mergeById(customers,r.data);root.innerHTML=customerRows(r.data)+pagers('customers',r.total)}catch(e){root.innerHTML='<div class="notice warn">客户分页加载失败：'+esc(e.message)+'</div>'}}
  function serverCustomersView(){state.customers=1;const html=pageHead('客户数据库','一客一档，客户级与商机级分开计算。服务端分页，数据量增大后手机仍只加载当前页。','<button class="btn btn-primary" onclick="customerModal()">＋ 新客户</button><button class="btn btn-soft" onclick="exportCustomers()">导出Excel兼容CSV</button>')+'<div class="card card-pad"><div class="filters"><input id="cq" placeholder="搜索公司/联系人/水果/市场"><select id="cg"><option value="">全部等级</option><option>S</option><option>A</option><option>B</option><option>C</option><option>D</option></select><select id="ct"><option value="">全部类型</option><option>货主</option><option>代卖</option><option>货主+代卖</option></select><select id="cc"><option value="">东/西货</option><option>东货</option><option>西货</option></select></div><div id="customerTable"></div></div>';document.querySelector('#view').innerHTML=html;let timer;['cq','cg','ct','cc'].forEach(id=>document.querySelector('#'+id).addEventListener('input',()=>{clearTimeout(timer);timer=setTimeout(()=>{state.customers=1;loadCustomers()},220)}));loadCustomers()}
  async function customerIdsFor(q){if(!q)return [];const or='(company_name.ilike.*'+q+'*,contact_name.ilike.*'+q+'*)';try{return (await ranged('/rest/v1/customers?archived_at=is.null&select=id&or='+encodeURIComponent(or)+'&limit=100')).data.map(x=>x.id)}catch{return []}}
  async function oppPath(){const parts=['archived_at=is.null','select=*','order=final_priority.desc','limit='+SIZE,'offset='+((state.opportunities-1)*SIZE)],q=val('oq').trim(),g=val('og'),a=val('oa'),s=val('os');if(q){const ids=await customerIdsFor(q),or=['fruit.ilike.*'+q+'*','current_port.ilike.*'+q+'*','origin_country.ilike.*'+q+'*'];if(ids.length)or.push('customer_id.in.('+ids.join(',')+')');parts.push('or='+encodeURIComponent('('+or.join(',')+')'))}if(g)parts.push('cargo_group=eq.'+encodeURIComponent(g));if(a)parts.push('alert_level=eq.'+encodeURIComponent(a));if(s)parts.push('stage=eq.'+encodeURIComponent(s));return '/rest/v1/opportunities?'+parts.join('&')}
  const F=(v,d=1)=>v===null||v===undefined||v===''?'-':Number(v).toLocaleString('zh-CN',{maximumFractionDigits:d}),MONEY=v=>v===null||v===undefined||v===''?'待补':'¥'+Number(v).toLocaleString('zh-CN',{maximumFractionDigits:2});
  function oppRows(rows){return '<div class="table-wrap"><table class="table"><thead><tr><th>客户/水果</th><th>阶段</th><th>JSOI</th><th>JGI</th><th>可信度</th><th>P50/P90差</th><th>ΔTCO</th><th>优先分</th><th>建议/操作</th></tr></thead><tbody>'+rows.map(o=>'<tr><td><span class="link" onclick="openCustomer(\''+o.customer_id+'\')">'+esc(customerName(o.customer_id))+'</span><div><b>'+esc(o.fruit)+'</b> · '+esc(o.current_transport||'')+'</div><div class="muted">'+esc(o.origin_country||'')+' '+esc(o.origin_region||'')+' → '+esc(o.current_port||'-')+' → '+esc(o.destination_market||'未填终点')+'</div></td><td><span class="stage">'+esc(o.stage)+'</span><div style="margin-top:6px">'+badge(o.alert_level)+'</div></td><td><b>'+F(o.jsoi_score)+'</b><div class="muted">统一切换指数</div></td><td><b>'+F(o.jgi_score)+'</b><div class="muted">江门短板</div></td><td>'+F(o.comparison_confidence,0)+'%</td><td>P50 '+F(o.delta_p50_days)+'天<div class="muted">P90 '+F(o.delta_p90_days)+'天</div></td><td>'+MONEY(o.delta_tco)+'</td><td><b>'+F(o.final_priority)+'</b></td><td>'+esc(o.client_recommendation||o.recommendation||'-')+'<div class="muted">短板：'+esc(o.jiangmen_gaps||'待补')+'</div><div style="margin-top:7px"><button class="btn btn-primary" onclick="routeCompareModal(\''+o.id+'\',false)">路线对比</button> <button class="btn btn-soft" onclick="routeCompareModal(\''+o.id+'\',true)">客户版</button> <button class="btn btn-soft" onclick="opportunityModal(\''+o.id+'\')">编辑</button></div></td></tr>').join('')+'</tbody></table></div>'}
  async function loadOpps(){const root=document.querySelector('#oppTable');if(!root)return;root.innerHTML='<div class="empty">正在读取商机第 '+state.opportunities+' 页…</div>';try{const r=await ranged(await oppPath());state.oppTotal=r.total;opportunities=mergeById(opportunities,r.data);const missing=[...new Set(r.data.map(o=>o.customer_id).filter(id=>!customers.some(c=>c.id===id)))];if(missing.length){try{const rr=await ranged('/rest/v1/customers?select=*&id=in.('+missing.join(',')+')&limit='+Math.min(100,missing.length));customers=mergeById(customers,rr.data)}catch{}}root.innerHTML=oppRows(r.data)+pagers('opportunities',r.total)}catch(e){root.innerHTML='<div class="notice warn">商机分页加载失败：'+esc(e.message)+'</div>'}}
  function serverOppsView(){state.opportunities=1;const html=pageHead('商机引擎','客户 × 水果 × 产区 × 月份 × 当前口岸，自动计算 OT / EPI / JSOI / JGI。服务端分页只加载当前页。','<button class="btn btn-primary" onclick="opportunityModal()">＋ 新商机</button><button class="btn btn-soft" onclick="exportOpps()">导出Excel兼容CSV</button>')+'<div class="card card-pad"><div class="filters"><input id="oq" placeholder="搜索客户/水果/口岸"><select id="og"><option value="">东/西货</option><option>东货</option><option>西货</option></select><select id="oa"><option value="">灯号</option><option>红</option><option>橙</option><option>黄</option><option>绿</option><option>蓝</option></select><select id="os"><option value="">阶段</option>'+['待验证','需求确认','方案','报价','试柜','持续合作','关闭'].map(x=>'<option>'+x+'</option>').join('')+'</select></div><div id="oppTable"></div></div>';document.querySelector('#view').innerHTML=html;let timer;['oq','og','oa','os'].forEach(id=>document.querySelector('#'+id).addEventListener('input',()=>{clearTimeout(timer);timer=setTimeout(()=>{state.opportunities=1;loadOpps()},220)}));loadOpps()}
  window.fonkonR11ServerPage=function(kind,delta){state[kind]=Math.max(1,(state[kind]||1)+delta);return kind==='customers'?loadCustomers():loadOpps()};
  try{customersView=serverCustomersView;window.customersView=serverCustomersView;oppsView=serverOppsView;window.oppsView=serverOppsView}catch{}
  window.fonkonR11ServerPaging=true;
})();
