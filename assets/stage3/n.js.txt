    if(!force&&remain>180)return ses;
    if(!ses.refresh_token)throw Error('登录会话需要重新登录');
    if(refreshPromise)return refreshPromise;
    const rt=ses.refresh_token;
    refreshPromise=(async()=>{
      const r=await fetch(U+'/auth/v1/token?grant_type=refresh_token',{method:'POST',headers:{apikey:K,'content-type':'application/json'},body:JSON.stringify({refresh_token:rt})});
      const t=await r.text();let j=null;try{j=t?JSON.parse(t):null}catch{j=null}
      if(!r.ok||!j?.access_token)throw Error(j?.error_description||j?.message||j?.error||'登录续期失败');
      saveSession(Object.assign({},ses,j));return ses;
    })().finally(()=>refreshPromise=null);
    return refreshPromise;
  }
