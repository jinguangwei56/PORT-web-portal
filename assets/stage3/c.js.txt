/* FONKON Stage3 C: KPI + reminders */
(function(){
function inPeriod(d,s,e){if(!d)return false;const x=new Date(d),a=new Date(s+'T00:00:00'),b=new Date(e+'T23:59:59');return x>=a&&x<=b}
function actualKpi(uid,t){const cs=customers.filter(c=>c.owner_id===uid&&inPeriod(c.created_at,t.period_start,t.period_end));const vs=visits.filter(v=>v.created_by===uid&&inPeriod(v.visited_at,t.period_start,t.period_end));const ops=opportunities.filter(o=>{const c=customers.find(c=>c.id===o.customer_id);return c?.owner_id===uid&&inPeriod(o.created_at,t.period_start,t.period_end)});const first=vs.filter(v=>v.visit_type==='首访').length;const valid=cs.length?Math.round(cs.filter(c=>Number(c.data_completeness)>=75&&Number(c.dc_score)>=50).length/cs.length*100):0;return {first_visits:first,valid_rate:valid,as_customers:cs.filter(c=>['A','S'].includes(c.cv_grade)).length,quotes:ops.filter(o=>['报价','试柜','持续合作'].includes(o.stage)).length,trial_ops:ops.filter(o=>Number(o.trial_volume)>0||['试柜','持续合作'].includes(o.stage)).length,trial_containers:ops.reduce((s,o)=>s+Number(o.trial_volume||0),0),won_ops:ops.filter(o=>o.stage==='持续合作').length,won_containers:ops.reduce((s,o)=>s+Number(o.won_volume||0),0)}}
const kpiDefs=[['first_visits','target_first_visits','首访数'],['valid_rate','target_valid_data_rate','有效数据率%'],['as_customers','target_as_customers','A/S客户'],['quotes','target_quotes','报价商机'],['trial_ops','target_trial_opportunities','试柜商机'],['trial_containers','target_trial_containers','试柜柜量'],['won_ops','target_won_opportunities','持续合作'],['won_containers','target_won_containers','成交柜量']];
function pct(a,b){if(!Number(b))return 0;return Math.min(999,Math.round(Number(a)/Number(b)*100))}
window.kpiView=async function(){try{const all=await req('/rest/v1/sales_kpi_targets?select=*&order=period_start.desc&limit=200');const now=new Date().toISOString().slice(0,10);const relevant=profile.role==='admin'?all:all.filter(t=>t.sales_user_id===ses.user.id);let html=pageHead('KPI目标管理','目标只衡量有效开发结果，不鼓励无意义电话量。',profile.role==='admin'?'<button class="btn btn-primary" onclick="kpiTargetModal()">＋设置目标</button>':'');if(!relevant.length){html+='<div class="card empty">尚未设置KPI目标</div>';return $('#view').innerHTML=html}const current=relevant.filter(t=>t.period_start<=now&&t.period_end>=now);const list=current.length?current:relevant.slice(0,10);html+='<div class="card card-pad">'+list.map(t=>{const a=actualKpi(t.sales_user_id,t),p=profiles.find(p=>p.id===t.sales_user_id)||profile;return '<div class="knowledge-block"><div class="section-title"><h2>'+esc(p?.name||'业务员')+' · '+t.period_start+' ~ '+t.period_end+'</h2><span class="hint">'+esc(t.notes||'')+'</span></div><div class="stat-grid">'+kpiDefs.map(([ak,tk,n])=>'<div class="stat-box"><b>'+fmt(a[ak])+' / '+fmt(t[tk])+'</b><span>'+n+' · '+pct(a[ak],t[tk])+'%</span>'+progress(pct(a[ak],t[tk]))+'</div>').join('')+'</div></div>'}).join('')+'</div>';$('#view').innerHTML=html}catch(e){toast(e.message)}};
window.kpiTargetModal=function(){if(profile.role!=='admin')return;const sales=(profiles||[]).filter(p=>p.role==='sales'&&p.active&&p.approval_status==='approved');if(!sales.length)return toast('暂无已启用业务员');const d=new Date(),start=new Date(d.getFullYear(),d.getMonth(),1).toISOString().slice(0,10),end=new Date(d.getFullYear(),d.getMonth()+1,0).toISOString().slice(0,10);modal('设置业务员KPI目标','<form id="kpiForm" class="grid"><div class="field span6"><label>业务员</label><select name="sales_user_id">'+sales.map(p=>'<option value="'+p.id+'">'+esc(p.name||p.email)+'</option>').join('')+'</select></div>'+input('period_start','开始日期',start,'date','span3')+input('period_end','结束日期',end,'date','span3')+input('target_first_visits','首访数',60,'number')+input('target_valid_data_rate','有效数据率%',70,'number')+input('target_as_customers','A/S客户',10,'number')+input('target_quotes','报价商机',8,'number')+input('target_trial_opportunities','试柜商机',4,'number')+input('target_trial_containers','试柜柜量',6,'number')+input('target_won_opportunities','持续合作客户',2,'number')+input('target_won_containers','成交柜量',10,'number')+input('notes','备注','','text','span12')+'<div class="span12"><button class="btn btn-primary">保存目标</button></div></form>');$('#kpiForm').onsubmit=async e=>{e.preventDefault();try{const x=Object.fromEntries(new FormData(e.target));['target_first_visits','target_valid_data_rate','target_as_customers','target_quotes','target_trial_opportunities','target_trial_containers','target_won_opportunities','target_won_containers'].forEach(k=>x[k]=Number(x[k]||0));x.created_by=ses.user.id;await req('/rest/v1/sales_kpi_targets',{method:'POST',headers:{Prefer:'resolution=merge-duplicates,return=minimal'},body:JSON.stringify(x)});closeModal();toast('KPI目标已保存');kpiView()}catch(z){toast(z.message)}}};

function reminderBase(){const now=Date.now(),out=[];tasks.filter(t=>t.status==='待办').forEach(t=>{if(t.due_at&&new Date(t.due_at).getTime()<=now)out.push({key:'task:'+t.id,level:'红',title:'跟进任务到期',text:customerName(t.customer_id)+' · '+t.title,view:'tasks'} )});opportunities.forEach(o=>{const nf=o.next_followup_at?new Date(o.next_followup_at).getTime():0;if(o.alert_level==='红'&&(!nf||nf<=now+48*3600000))out.push({key:'red:'+o.id,level:'红',title:'红灯商机48小时动作',text:customerName(o.customer_id)+' · '+o.fruit+' · '+(o.recommendation||''),view:'opportunities'});if(o.season_window_days!=null&&o.season_window_days>0&&o.season_window_days<=28)out.push({key:'season:'+o.id+':'+o.season_window_days,level:o.season_window_days<=14?'橙':'黄',title:'产季窗口临近',text:customerName(o.customer_id)+' · '+o.fruit+' · 距高峰约 '+o.season_window_days+' 天',view:'opportunities'});if(o.target_trial_date){const dd=Math.ceil((new Date(o.target_trial_date+'T00:00:00').getTime()-now)/86400000);if(dd>=0&&dd<=7)out.push({key:'trial:'+o.id+':'+dd,level:'橙',title:'试柜日期临近',text:customerName(o.customer_id)+' · '+o.fruit+' · '+dd+'天后试柜',view:'opportunities'})}});if(profile.role==='admin'){(profiles||[]).filter(p=>p.role==='sales'&&p.approval_status==='pending').forEach(p=>out.push({key:'approve:'+p.id,level:'橙',title:'业务员注册待审批',text:p.name+' · '+(p.email||''),view:'team'}))}return out}
async function reminderFeed(){let out=reminderBase();if(profile.role==='admin'){try{const [ks,mi]=await Promise.all([req('/rest/v1/knowledge_submissions?review_status=eq.%E5%BE%85%E5%AE%A1%E6%A0%B8&select=id,title,fruit'),req('/rest/v1/market_intel?review_status=eq.%E5%BE%85%E5%AE%A1%E6%A0%B8&select=id,fruit,finding')]);ks.forEach(x=>out.push({key:'ks:'+x.id,level:'黄',title:'知识补充待审核',text:(x.fruit||'')+' · '+(x.title||''),view:'knowledge'}));mi.forEach(x=>out.push({key:'mi:'+x.id,level:'黄',title:'市场情报待审核',text:(x.fruit||'')+' · '+(x.finding||''),view:'knowledge'}))}catch{}}return out}
async function notificationStates(){try{return await req('/rest/v1/notification_state?select=*')}catch{return []}}
window.remindersView=async function(){const [feed,st]=await Promise.all([reminderFeed(),notificationStates()]);const dismissed=new Set(st.filter(x=>x.dismissed_at).map(x=>x.notification_key));const arr=feed.filter(x=>!dismissed.has(x.key));let html=pageHead('提醒中心','把到期跟进、红灯商机、产季窗口、试柜与审批集中在一个地方。','<button class="btn btn-soft" onclick="markAllReminderRead()">全部已读</button>');html+='<div class="card card-pad">'+(arr.length?arr.map(r=>'<div class="task-card"><div class="task-top"><div><b>'+esc(r.title)+'</b> '+badge(r.level)+'<div class="task-meta">'+esc(r.text)+'</div></div><div class="actions"><button class="btn btn-primary" onclick="render(\''+r.view+'\')">处理</button><button class="btn btn-soft" onclick="dismissReminder(\''+r.key.replace(/'/g,"\\'")+'\')">忽略</button></div></div></div>').join(''):'<div class="empty">当前没有待处理提醒</div>')+'</div>';$('#view').innerHTML=html;await updateReminderBell()};
window.dismissReminder=async key=>{try{await req('/rest/v1/notification_state',{method:'POST',headers:{Prefer:'resolution=merge-duplicates,return=minimal'},body:JSON.stringify({user_id:ses.user.id,notification_key:key,dismissed_at:new Date().toISOString(),updated_at:new Date().toISOString()})});remindersView()}catch(e){toast(e.message)}};
window.markAllReminderRead=async()=>{const feed=await reminderFeed();for(const r of feed){try{await req('/rest/v1/notification_state',{method:'POST',headers:{Prefer:'resolution=merge-duplicates,return=minimal'},body:JSON.stringify({user_id:ses.user.id,notification_key:r.key,read_at:new Date().toISOString(),updated_at:new Date().toISOString()})})}catch{}}toast('已标记已读');updateReminderBell()};
window.updateReminderBell=async function(){if(!profile)return;const [feed,st]=await Promise.all([reminderFeed(),notificationStates()]);const hidden=new Set(st.filter(x=>x.dismissed_at).map(x=>x.notification_key)),read=new Set(st.filter(x=>x.read_at).map(x=>x.notification_key));const n=feed.filter(x=>!hidden.has(x.key)&&!read.has(x.key)).length;let b=document.querySelector('#reminderBell');if(!b){b=document.createElement('button');b.id='reminderBell';b.className='btn btn-soft';b.onclick=()=>render('reminders');const who=document.querySelector('#who');who?.parentElement?.insertBefore(b,who)}b.innerHTML='🔔'+(n?'<b style="margin-left:4px;color:#dc2626">'+n+'</b>':'')};
const prevAfter=window.fonkonAfterRefresh;window.fonkonAfterRefresh=async function(){if(prevAfter)await prevAfter();await updateReminderBell()};
const oldRender=render;render=function(v){if(v==='kpi')return kpiView();if(v==='reminders')return remindersView();return oldRender(v)};
setTimeout(()=>{if(profile)updateReminderBell()},800);
})();
