window.preVisitAssistantView=async function(fruitPreset=''){

 const cs='<option value="">不绑定现有客户</option>'+customers.map(c=>'<option value="'+c.id+'">'+esc(c.company_name)+'</option>').join('');

 const target=$('#knowledgeBody')||$('#view');
 target.innerHTML='<div class="assistant-card"><div class="section-title"><h2>⚡ 拜访前3分钟作战助手</h2><span class="hint">把公司知识自动转成今天要问的问题</span></div><form id="assistantForm" class="grid"><div class="field span6"><label>客户</label><select name="customer_id">'+cs+'</select></div><div class="field span6"><label>本次主谈水果</label><select name="fruit">'+fo+'</select></div><div class="span12"><button class="btn btn-primary">生成拜访作战卡</button></div></form></div><div id="assistantResult"></div>';
 $('#assistantForm').onsubmit=async e=>{e.preventDefault();const x=Object.fromEntries(new FormData(e.target));await generatePreVisitCard(x.customer_id,x.fruit)};
 if(fruitPreset)await generatePreVisitCard('',fruitPreset);

window.generatePreVisitCard=async function(cid,fruit){
 const c=customers.find(x=>x.id===cid)||{}; const group=(await req('/rest/v1/fruit_catalog?fruit=eq.'+encodeURIComponent(fruit)+'&select=cargo_group'))[0]?.cargo_group||'';
 const [orig,clr,cold,intel]=await Promise.all([req('/rest/v1/fruit_origin_knowledge?fruit=eq.'+encodeURIComponent(fruit)+'&select=*&order=evidence_level.asc&limit=20'),req('/rest/v1/fruit_clearance_knowledge?fruit=eq.'+encodeURIComponent(fruit)+'&select=*&limit=30'),req('/rest/v1/fruit_coldchain_knowledge?fruit=eq.'+encodeURIComponent(fruit)+'&select=*&limit=10'),req('/rest/v1/market_intel?fruit=eq.'+encodeURIComponent(fruit)+'&review_status=eq.'+encodeURIComponent('已采纳')+'&select=*&order=collected_at.desc&limit=10')]);
 const regions=orig.slice(0,5).map(x=>[x.country,x.region].filter(Boolean).join('·')).join('、')||'公司尚未录入已审核产区';
 const peaks=orig.slice(0,4).map(x=>x.country+' '+(x.peak_start_month||'-')+'–'+(x.peak_end_month||'-')+'月').join('；')||'待补';
 const items=[...new Set(clr.map(x=>x.item_name).filter(Boolean))].slice(0,8).join('、')||'待核验';
 const tat=clr.filter(x=>x.typical_min_days!=null||x.typical_max_days!=null).slice(0,3).map(x=>x.item_name+' '+(x.typical_min_days??'-')+'–'+(x.typical_max_days??'-')+'天').join('；')||'不得写死，按实际口岸/项目核验';
 const risks=cold.map(x=>x.common_damage_risks).filter(Boolean).slice(0,3).join('；')||'待录入';
 const recent=intel.slice(0,3).map(x=>x.finding).filter(Boolean).join('；')||'暂无已验证近期市场情报';
 const qs=[
 c.decision_role?('确认：本次路线/口岸最终还是由“'+c.decision_role+'”决定吗？'):'谁是真正决定运输路线、口岸和清关服务的人？',
 c.annual_containers?('你们这个水果全年约'+fmt(c.annual_containers)+'柜，其中旺季每月大概多少柜？'):'这个水果全年多少柜？旺季单月最高多少柜？',
 '主要来自哪个国家、哪个产区？采摘和集中进口高峰分别是哪几个月？',
 '目前主要走哪条路线、哪个口岸？淡季从装货到可提货实际几天？',
 '旺季最堵时实际几天、排队多久？请给最近一次真实案例。',
 '最近一批货有多少柜被现场查验？其中多少柜被抽样送检？具体检测了什么、等了几天？',
 '这些延误/查验/货损实际给你造成多少元/柜或多少天的损失？',
 '如果江门方案在旺季能给出可验证的时效/风险优势，即使成本存在差异，你愿意拿1–3柜做试柜验证吗？'
 ];
 $('#assistantResult').innerHTML='<div class="two section"><div class="knowledge-block"><h3>'+esc(fruit)+'｜'+esc(group)+'｜3分钟速读</h3><div class="knowledge-row"><b>主要已审核产区：</b>'+esc(regions)+'</div><div class="knowledge-row"><b>集中上市：</b>'+esc(peaks)+'</div><div class="knowledge-row"><b>已审核查验/送检知识：</b>'+esc(items)+'</div><div class="knowledge-row"><b>参考送检时效：</b>'+esc(tat)+'</div><div class="knowledge-row"><b>冷链/品质风险：</b>'+esc(risks)+'</div><div class="knowledge-row"><b>近期共享市场情报：</b>'+esc(recent)+'</div><div class="notice">凡显示“待核验/待录入”的内容，业务员不得当作确定事实向客户承诺。</div></div><div class="knowledge-block"><h3>今天必须问的8个问题</h3><div class="question-list">'+qs.map(q=>'<div>'+esc(q)+'</div>').join('')+'</div></div></div>';
}
